### Q3. 이름이 홍길동인 고객의 1번 계좌의 과거 한달간 입출금 내역을 출력하라
```sql
WITH hong_account AS (
  SELECT a.account_id,
         ROW_NUMBER() OVER (PARTITION BY a.cust_ssn
                            ORDER BY a.open_date, a.account_id) AS account_seq
  FROM customer c
  JOIN account  a ON a.cust_ssn = c.cust_ssn
  WHERE c.cust_name = '홍길동'
)
SELECT t.account_id    AS 계좌번호,
       t.txn_date      AS 거래일자,
       t.txn_time      AS 거래시각,
       t.txn_no        AS 거래번호,
       t.txn_type      AS 구분,
       t.txn_desc      AS 내용,
       t.amount        AS 거래금액,
       t.balance_after AS 거래후잔액,
       t.channel       AS 채널
FROM hong_account h
JOIN account_txn  t ON t.account_id = h.account_id
WHERE h.account_seq = 1
  AND t.txn_date BETWEEN CURDATE() - INTERVAL 1 MONTH AND CURDATE()
ORDER BY t.txn_date, t.txn_no;
```

### Q4. 홍길동 고객이 보유한 모든 신용 카드와 연결 계좌, 현재까지의 사용 금액 및 사용 한도를 출력하라
```sql
SELECT k.card_id       AS 카드번호,
       k.account_id    AS 연결계좌,
       a.account_type  AS 연결계좌종류,
       k.credit_limit  AS 사용한도,
       COALESCE(SUM(CASE WHEN u.status = '승인' THEN u.amount END), 0) AS 사용금액,
       k.credit_limit
         - COALESCE(SUM(CASE WHEN u.status = '승인' THEN u.amount END), 0) AS 잔여한도
FROM customer   c
JOIN card       k ON k.cust_ssn   = c.cust_ssn
JOIN account    a ON a.account_id = k.account_id
LEFT JOIN card_usage u ON u.card_id = k.card_id
WHERE c.cust_name = '홍길동'
  AND k.card_type = '신용'
GROUP BY k.card_id, k.account_id, a.account_type, k.credit_limit
ORDER BY k.card_id;
```

### Q5. 현재 날짜에서 가장 생일이 먼저 다가오는 고객의 이름, 주소 및 생일은?
```sql
WITH birthday AS (
  SELECT cust_name, cust_addr, cust_birth,
         cust_birth + INTERVAL
           (TIMESTAMPDIFF(YEAR, cust_birth, CURDATE() - INTERVAL 1 DAY) + 1) YEAR
           AS next_birthday
  FROM customer
)
SELECT cust_name                           AS 고객명,
       cust_addr                           AS 주소,
       cust_birth                          AS 생년월일,
       next_birthday                       AS 다가오는생일,
       DATEDIFF(next_birthday, CURDATE())  AS 남은일수
FROM birthday
WHERE next_birthday = (SELECT MIN(next_birthday) FROM birthday);
```

```text
| 최민호 | 서울특별시 종로구 삼일대로 30 | 1985-11-30 | 2026-11-30   |       25 |
```
